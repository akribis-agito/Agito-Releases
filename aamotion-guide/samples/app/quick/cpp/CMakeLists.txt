cmake_minimum_required(VERSION 3.20)
project(AAMotionQuickExamples CXX)

set(AAMOTION_SDK_DIR "" CACHE PATH "Extracted AAMotion Native SDK")
if(NOT AAMOTION_SDK_DIR OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Set AAMOTION_SDK_DIR and configure a 64-bit build")
endif()

set(CMAKE_CXX_STANDARD 17)
if(WIN32)
  set(runtime win-x64)
  set(library aamotion.lib)
  set(runtime_files aamotion.dll AACommServerNative.exe)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(runtime linux-x64)
  set(library libaamotion.so)
  set(runtime_files libaamotion.so AACommServerNative)
  set(CMAKE_BUILD_RPATH "$ORIGIN")
else()
  message(FATAL_ERROR "These projects target Windows x64 and Linux x64")
endif()

foreach(example ReadInput MoveTwoAxes HomeAxis TwoSpeed CncLine DataLogger SlowLogger)
  add_executable(${example} ${example}.cpp)
  set_target_properties(${example} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${example}")
  target_include_directories(${example} PRIVATE "${AAMOTION_SDK_DIR}/include")
  if(MSVC)
    target_compile_options(${example} PRIVATE /utf-8)
  endif()
  target_link_libraries(${example} PRIVATE "${AAMOTION_SDK_DIR}/${runtime}/${library}")
  foreach(file IN LISTS runtime_files)
    add_custom_command(TARGET ${example} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${AAMOTION_SDK_DIR}/${runtime}/${file}" "$<TARGET_FILE_DIR:${example}>")
  endforeach()
endforeach()

    同时控制多台控制器¶

    每台控制器一个控制器对象（C 中是一个句柄），型号各自指定。所有对象共用本机的同一个 AACommServer，服务器为每台控制器维护各自的连接和队列，所以两台控制器的命令互不排队。

    这两个完整工程并行连接两台 AGM800，并分别读取 A 轴位置。编译后以两个不同的控制器 IP 为参数运行。其他型号请修改初始化时的型号。

    using System;
using System.Threading.Tasks;
using AAMotion;

internal static class Program
{
    private static int Main(string[] args)
    {
        if (args.Length != 2 || args[0] == args[1])
        {
            Console.Error.WriteLine("Usage: MultiController <first-ip> <second-ip>");
            return 1;
        }

        Task<bool> first = Task.Run(() => ReadPosition(args[0]));
        Task<bool> second = Task.Run(() => ReadPosition(args[1]));
        Task.WaitAll(first, second);
        return first.Result && second.Result ? 0 : 2;
    }

    private static bool ReadPosition(string ip)
    {
        MotionController controller = AAMotionAPI.Initialize(ControllerType.AGM800);
        try
        {
            if (!AAMotionAPI.Connect(controller, ip))
            {
                Console.Error.WriteLine($"{ip}: connection failed");
                return false;
            }

            Console.WriteLine($"{ip}: APos={controller.GetAxis(AxisRef.A).Pos}");
            return true;
        }
        catch (ControllerCommandException error)
        {
            Console.Error.WriteLine($"{ip}: {error.Command} ERR {error.ErrorCode}");
            return false;
        }
        finally
        {
            AAMotionAPI.Disconnect(controller);
        }
    }
}

    #include "aamotion.h"
#include <cstdio>
#include <cstring>
#include <future>

static bool ReadPosition(const char* ip)
{
    auto controller = aamotion_initialize(AAMOTION_CONTROLLER_TYPE_AGM800);
    if (!controller)
    {
        std::fprintf(stderr, "%s: initialization failed\n", ip);
        return false;
    }

    double position = 0;
    bool ok = aamotion_connect(controller, ip, false, false) &&
        aamotion_get_value(controller, AAMOTION_AXIS_REF_A,
                          AAMOTION_KEYWORD_POS, 0, &position);
    if (ok)
        std::printf("%s: APos=%g\n", ip, position);
    else
        std::fprintf(stderr, "%s: connect or read failed\n", ip);

    aamotion_disconnect(controller);
    return ok;
}

int main(int argc, char** argv)
{
    if (argc != 3 || std::strcmp(argv[1], argv[2]) == 0)
    {
        std::fprintf(stderr, "Usage: multi_controller <first-ip> <second-ip>\n");
        return 1;
    }

    auto first = std::async(std::launch::async, ReadPosition, argv[1]);
    auto second = std::async(std::launch::async, ReadPosition, argv[2]);
    const bool firstOk = first.get();
    const bool secondOk = second.get();
    return firstOk && secondOk ? 0 : 2;
}

    要点¶

    • 并行连接：每次连接可能要几秒，两台一起连（Task.Run）比依次连快。
    • 各自的超时和错误钩子：SetDefaultTimeout、ErrorOccurred 都是按对象设置的。
    • 同一台控制器不要创建两个对象：两个对象会建立两个连接，互相干扰。程序里共享同一个对象。
    • IP 地址：每台控制器的 IP 要不同，电脑网卡要在它们的网段内。

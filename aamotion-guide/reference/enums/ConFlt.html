    控制器故障码 ConFlt¶

    关键字 ConFlt：最近一次导致断使能的故障。0 表示没有故障。故障排除后，重新使能会清除它。
    C# 类型 AAMotion.ConFlt
    C 类型 aamotion_con_flt_t
    转成文字 EnumPropertyHelper.GetConFltDes(int) · aamotion_get_con_flt_des()

    读取并解释¶

    int v = (int)c.GetValue(AxisRef.A, eKeyword.ConFlt);
ConFlt e = (ConFlt)v;
string text = EnumPropertyHelper.GetConFltDes(v);

    double v;
if (aamotion_get_value(h, AAMOTION_AXIS_REF_A, AAMOTION_KEYWORD_CON_FLT, 0, &v)) {
    char text[256];
    aamotion_get_con_flt_des((int)v, text, sizeof text);
    printf("%d: %s\n", (int)v, text);
}

    v = int(am.get_value(h, am.AXIS_A, am.KEYWORD_CON_FLT))
print(v, am.describe("con_flt", v))

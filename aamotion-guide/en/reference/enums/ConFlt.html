    Controller fault codes ConFlt¶

    Keyword ConFlt, the last fault that turned the motor off. 0 means no fault. It clears when the motor is enabled again after the fault is removed.
    C# type AAMotion.ConFlt
    C type aamotion_con_flt_t
    To text EnumPropertyHelper.GetConFltDes(int) · aamotion_get_con_flt_des()

    Read and decode¶

    int v = (int)c.GetValue(AxisRef.A, eKeyword.ConFlt);
ConFlt e = (ConFlt)v;
string text = EnumPropertyHelper.GetConFltDes(v);

    double v;
if (aamotion_get_value(h, AAMOTION_AXIS_REF_A, AAMOTION_KEYWORD_CON_FLT, 0, &v)) {
    char text[256];
    aamotion_get_con_flt_des((int)v, text, sizeof text);
    printf("%d: %s\n", (int)v, text);
}

    v = int(am.get_value(h, am.AXIS_A, am.KEYWORD_CON_FLT))
print(v, am.describe("con_flt", v))

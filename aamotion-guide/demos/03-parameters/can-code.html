    CAN 编号与参数类型¶

    从已连接控制器的命令表读关键字的 CAN 编号和数据类型。CAN 编号随产品和固件不同，eKeyword 的数值只是 SDK 内部标识，不能代替 CAN 编号；慢速记录器、ECAM 主轴等配置用这里读出的值。

    1. 连接
    2. 本次操作：CanCode
    3. 断开

    C# samples/single/csharp/03-Parameters/CanCode

    PCSuite 中对应：参数管理

    运行前提¶

    AGM800；轴不运动。默认连接控制器 172.1.1.101；没有设备时把 Target 改成仿真器 firmware_sim.exe 的完整路径。

    核心代码¶

    // 2. 本次操作
// CAN 编号随产品和固件而变
// eKeyword 的数值不能当作 CAN 编号
int code = c.GetKeywordCANCode(eKeyword.Pos);
int element =
    c.GetKeywordCANCode(eKeyword.LoggerParams, AxisRef.A, 1);
string type = c.GetParameterType(eKeyword.Pos);
Console.WriteLine($"Pos: CAN {code}, {type}");
Console.WriteLine("ALoggerParams[1]: CAN " + element);

    C++ / Python

    这个功能是 C# 专有的写法，C 接口没有对应函数，所以只有 C# 版本。

    如何确认结果¶

    输出 Pos 的 CAN 编号与类型，以及 ALoggerParams[1] 的 CAN 编号。

    退出码：0 成功；1 未确认或不适用；2 连接失败；3 换相超时；4 使能被拒绝；5 命令失败或异常；6 等待或结果检查失败。

    运行¶

    1. 用 Visual Studio 2019 或更新版本打开 samples/single/csharp/AAMotionDemos.sln，也可以只打开本工程的 CanCode.csproj。
    2. 打开 Program.cs，确认顶部的 Target：默认是控制器出厂 IP 172.1.1.101；用仿真器时可将 firmware_sim.exe 的绝对路径作为第一个调试参数。
    3. 在解决方案资源管理器中右键 CanCode →“设为启动项目”，按 F5 运行。
    4. 需要 .NET Framework 4.8 开发工具；首次生成会自动还原 NuGet 包 Agito.AAMotion 13.0.0。

    对应 API：CAN 编号与参数类型

    using System;
using System.IO;
using AAMotion;
// CAN 编号与参数类型 / CAN code and type
// 顺序：连接 → 本次操作 → 断开。轴不运动。
internal static class Program
{
    // 控制器 IP（出厂默认 172.1.1.101）；用仿真器时改成 firmware_sim.exe 的完整路径
    private const string Target = "172.1.1.101";

    private static int Main(string[] args)
    {
        string connectionTarget = args.Length == 0 ? Target : args[0];
        MotionController c = AAMotionAPI.Initialize(ControllerType.AGM800);
        c.ErrorOccurred += (code, command, message) =>
            Console.WriteLine($"[ERR] {code}: {command} | {message}");
        try
        {
            // 1. 连接：Target 是 IP 时连接控制器；以 .exe 结尾时连接仿真器
            Console.WriteLine(connectionTarget);
            bool connected;
            bool simulation = connectionTarget.EndsWith(".exe",
                StringComparison.OrdinalIgnoreCase);
            if (!simulation)
                connected = AAMotionAPI.Connect(c, connectionTarget);
            else if (File.Exists(connectionTarget))
            {
                c.Init();   // 仿真器需要先启动 AACommServer
                connected = AAMotionAPI.ConnectSimulation(c, connectionTarget);
            }
            else
            {
                Console.WriteLine("simulator not found; edit Target");
                return 2;
            }
            if (!connected)
            {
                Console.WriteLine("failed; check Target and the cable");
                return 2;
            }

            // 2. 本次操作
            // CAN 编号随产品和固件而变
            // eKeyword 的数值不能当作 CAN 编号
            int code = c.GetKeywordCANCode(eKeyword.Pos);
            int element =
                c.GetKeywordCANCode(eKeyword.LoggerParams, AxisRef.A, 1);
            string type = c.GetParameterType(eKeyword.Pos);
            Console.WriteLine($"Pos: CAN {code}, {type}");
            Console.WriteLine("ALoggerParams[1]: CAN " + element);
            return 0;
        }
        catch (ControllerCommandException ex)
        {
            Console.WriteLine($"{ex.Command} refused: ERR {ex.ErrorCode}");
            return 5;
        }
        catch (Exception ex)
        {
            Console.WriteLine(ex.Message);
            return 5;
        }
        finally
        {
            // 3. 断开：AACommServer 继续为其他程序服务
            AAMotionAPI.Disconnect(c);
            Console.WriteLine("done");
        }
    }
}

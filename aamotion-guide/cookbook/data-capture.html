    采集运动曲线¶

    选择记录器¶

    记录器 最短采样间隔 适合记录
    SlowLogger 约 1 ms（LoggerGap = 1） 长时间趋势，例如连续记录位置误差或温度。运行中由上位机定期上传 FIFO 数据。
    DataLogger 一个控制器扫描周期，典型约 61 µs 单次运动、启动瞬态等短时波形。数据先保存在控制器内存，采集结束后上传。

    DataLogger 示例在 A 轴开始运动时触发，记录 APos 和 APosErr。示例参数为 1 秒、请求间隔 1 ms、触发前保留 10%。C# 和 C++ 都输出 CSV。需要增加信号时，先用 GetDataRecordingCapacity 或 aamotion_data_rec_capacity 查询控制器容量。

    旧手册的八个采集场景¶

    场景 操作顺序和检查项
    一键录制 已知时长和通道时用 RecordData。成功后检查每通道点数与采样间隔，再保存 CSV。
    先查容量 调 GetDataRecordingCapacity。同时比较通道上限和总样本容量。粗略估算：时长 2 s、间隔 1 ms、3 通道需要约 6000 个样本槽位，以实际下发的点数为准。
    在某位置触发 设 APos 为触发源，阈值使用该轴当前用户单位。正向跨过阈值用上升沿，反向用下降沿。开动前确认已经进入状态 2。
    记录运动起止 用 AMotionStat 的启动条件触发，保留预触发样本并给足记录时长。需要单独观察停下时，配置匹配的下降沿或在记录中保留状态通道。
    分步观察进度 ConfigureDataLogger → StartDataRecording → 查询 RecStat 和计数。ReadDataLoggerIncrementalData 返回增量计数，实际波形通过上传获得。
    多控制器分包上传 每台控制器独立设置和等待。支持 RecUploadNext 时可以用 UploadDataLoggerDataNonblocking 分包传输，该调用仍等待返回。
    异步上传给 UI 使用 StartUploadDataLoggerDataNonblocking 注册完成、进度和错误回调。让 UI 用自己的调度机制刷新。回调里保持短操作和快速返回。
    强制触发自查 在轴静止时准备好记录器，确认已进入状态 2，再用 ForceDataLoggerTrigger 验证触发、等待和上传链路。

    触发没出现时，先核对状态 1 或 2，再查信号、触发沿、掩码和阈值单位。采集结束后用状态 4 判断是否正常完成，状态 5 或 6 表示被停止。分包上传不可用时改用固件支持的普通上传方式。

    强制触发验证（C#）¶

    以下示例接收已连接的控制器，保持 A 轴静止，用强制触发验证记录器链路。

    var config = new MotionController.DataLoggerConfiguration
{
    DurationSec = 0.25,
    GapMs = 1,
    TriggerPosition = 10,
    Trigger = new MotionController.DataLoggerTrigger
    {
        Type = 5, Source = "AMotionStat", Value = 0, Mask = -1,
    },
};
config.ParamsToRecord.Add("APos");

string error = c.ConfigureDataLogger(config);
if (string.IsNullOrEmpty(error)) error = c.StartDataRecording();
if (!string.IsNullOrEmpty(error)) { Console.WriteLine(error); return; }

try
{
    var timer = Stopwatch.StartNew();
    while (timer.ElapsedMilliseconds < 3000)
    {
        error = c.GetDataRecordingStatus(out int status);
        if (!string.IsNullOrEmpty(error)) { Console.WriteLine(error); return; }
        if (status == 2) break;
        System.Threading.Thread.Sleep(10);
    }
    if (timer.ElapsedMilliseconds >= 3000) { Console.WriteLine("Recorder did not arm."); return; }
    error = c.ForceDataLoggerTrigger();
    if (string.IsNullOrEmpty(error)) error = c.WaitForDataRecordingToEnd(3000);
    if (!string.IsNullOrEmpty(error)) { Console.WriteLine(error); return; }
    error = c.GetDataRecordingStatus(out int finalStatus);
    if (!string.IsNullOrEmpty(error)) { Console.WriteLine(error); return; }
    if (finalStatus != 4) { Console.WriteLine($"Recording ended in state {finalStatus}."); return; }
    error = c.UploadDataLoggerData(out MotionController.DataLoggerResult result);
    Console.WriteLine(string.IsNullOrEmpty(error) ? $"{result.Vectors[0].Values.Count} samples" : error);
}
finally
{
    c.StopDataRecording();
}

    在 PCSuite 中核对采集配置¶

    完整示例¶

    要在 PCSuite 验证同一条曲线，在 Scope 或记录工具里选 A 轴位置 APos 和跟随误差 APosErr，把触发源设为 AMotionStat，条件设为从 0 向上跨过阈值。请求记录时长 1 秒、间隔 1 ms、预触发 10%。等记录器进入等待触发再执行同一次运动。导出后先对照通道、触发点和实际间隔，再比较波形。PCSuite 的表达式曲线由原始通道计算得到，配置 SDK 记录参数时填表达式使用的控制器通道。

    只在直接解析 AAComm 原始上传时处理录制头。固件可能使用 V1、V2 或 V3，头长度及字段位置会变化。普通 AAMotion 程序使用 SDK 解析出的通道与采样数据。

    C# 打开 samples/single/csharp/AAMotionDemos.sln 并选择对应工程。C++ 从 samples/single/cpp 配置 CMake，设置 AAMOTION_SDK_DIR，再生成对应目标。运行前修改源码顶部的 Target。

    DataLogger：采集一次运动¶

    using System;
using System.Diagnostics;
using System.IO;
using System.Threading;
using AAMotion;
// 记录一次运动 / Record a move
// 顺序：连接 → 换相 → 使能 → 本次操作 → 确认结果 → 清理与断开
internal static class Program
{
    // 控制器 IP（出厂默认 172.1.1.101）；用仿真器时改成 firmware_sim.exe 的完整路径
    private const string Target = "172.1.1.101";

    private static int Main(string[] args)
    {
        Console.WriteLine("Axis A will move. Check travel and units.");
        Console.WriteLine("Type RUN to continue:");
        if (Console.ReadLine() != "RUN") return 1;
        string connectionTarget = args.Length == 0 ? Target : args[0];
        MotionController c = AAMotionAPI.Initialize(ControllerType.AGM800);
        c.ErrorOccurred += (code, command, message) =>
            Console.WriteLine($"{code}: {command} | {message}");
        bool motorOnRequested = false;
        try
        {
            // 1. 连接：Target 是 IP 时连接控制器；以 .exe 结尾时连接仿真器
            Console.WriteLine(connectionTarget);
            bool connected;
            bool simulation = connectionTarget.EndsWith(".exe",
                StringComparison.OrdinalIgnoreCase);
            if (!simulation)
                connected = AAMotionAPI.Connect(c, connectionTarget);
            else if (File.Exists(connectionTarget))
            {
                c.Init();   // 仿真器需要先启动 AACommServer
                connected = AAMotionAPI.ConnectSimulation(c, connectionTarget);
            }
            else
            {
                Console.WriteLine("simulator not found; edit Target");
                return 2;
            }
            if (!connected)
            {
                Console.WriteLine("failed; check Target and the cable");
                return 2;
            }
            ISingleAxis a = c.GetAxis(AxisRef.A);

            // 仿真用虚拟电机。真实电机才需要换相。
            if (simulation)
                a.MotorType = 5;
            else if (!AAMotionAPI.IsAutoPhase(c, AxisRef.A))
            {
                Console.WriteLine("requested");
                AAMotionAPI.AutoPhase(c, AxisRef.A);
                var phaseTime = Stopwatch.StartNew();
                while (!AAMotionAPI.IsAutoPhase(c, AxisRef.A))
                {
                    if (phaseTime.ElapsedMilliseconds >= 10000)
                    {
                        Console.WriteLine("timed out; motor stays off");
                        return 3;
                    }
                    Thread.Sleep(20);
                }
            }
            if (!simulation) Console.WriteLine("commutated");

            // 3. 使能：清理阶段也会处理结果不确定的使能
            motorOnRequested = true;
            AAMotionAPI.MotorOn(c, AxisRef.A);
            if (a.MotorOn != 1)
            {
                Console.WriteLine("refused; MotorOn=" + a.MotorOn);
                return 4;
            }
            Console.WriteLine("on");

            // 4. 本次操作
            var cfg = new MotionController.DataLoggerConfiguration
            {
                DurationSec = 1,       // 记录 1 秒
                GapMs = 1,             // 每 1 ms 一个点
                TriggerPosition = 10,  // 触发前保留 10 %
                // 类型 5：AMotionStat 从 0 升为非零，即 A 轴开始运动
                Trigger = new MotionController.DataLoggerTrigger
                {
                    Type = 5, Source = "AMotionStat", Value = 0, Mask = -1,
                },
            };
            cfg.ParamsToRecord.Add("APos");
            cfg.ParamsToRecord.Add("APosErr");
            string error = c.ConfigureDataLogger(cfg);
            if (!string.IsNullOrEmpty(error))
                throw new InvalidOperationException(error);

            MotionController.DataLoggerResult result;
            try
            {
                error = c.StartDataRecording();
                if (!string.IsNullOrEmpty(error))
                    throw new InvalidOperationException(error);

                // 先填满预触发缓冲区，再等待运动开始信号。
                var armTime = Stopwatch.StartNew();
                int state;
                do
                {
                    error = c.GetDataRecordingStatus(out state);
                    if (!string.IsNullOrEmpty(error))
                        throw new InvalidOperationException(error);
                    if (state == 2) break;
                    if (armTime.ElapsedMilliseconds >= 2000)
                        throw new TimeoutException("Recorder is not armed.");
                    Thread.Sleep(10);
                } while (true);

                a.MoveRel(5000, speed: 5000, accel: 50000, decel: 50000);
                error = c.WaitForDataRecordingToEnd(5000);
                if (!string.IsNullOrEmpty(error))
                    throw new InvalidOperationException(error);
                error = c.UploadDataLoggerData(out result);
                if (!string.IsNullOrEmpty(error))
                    throw new InvalidOperationException(error);
            }
            finally
            {
                c.StopDataRecording();
            }
            foreach (MotionController.DataLoggerVector v in result.Vectors)
                Console.WriteLine($"{v.Parameter}: {v.Values.Count} samples");
            string csv = MotionController.GenerateCsvFile(result,
                Environment.CurrentDirectory);
            Console.WriteLine("CSV: " + csv);

            // 5. 确认结果：方法返回不等于动作完成
            var motionTime = Stopwatch.StartNew();
            while (a.InTargetStat != 4)
            {
                if (motionTime.ElapsedMilliseconds >= 10000)
                {
                    AAMotionAPI.Stop(c, AxisRef.A);
                    Console.WriteLine("settle timeout; Stop requested");
                    return 6;
                }
                Thread.Sleep(20);
            }
            Console.WriteLine($"settled: Pos={a.Pos}, " +
                $"MotionStat={a.MotionStat}, InTargetStat={a.InTargetStat}");
            return 0;
        }
        catch (ControllerCommandException ex)
        {
            Console.WriteLine($"{ex.Command} refused: ERR {ex.ErrorCode}");
            return 5;
        }
        catch (Exception ex)
        {
            Console.WriteLine(ex.Message);
            return 5;
        }
        finally
        {
            // 6. 清理：先确认停止再断使能
            //    断开通信不能证明设备已停止
            try
            {
                if (motorOnRequested)
                {
                    AAMotionAPI.Stop(c, AxisRef.A);
                    var stopTime = Stopwatch.StartNew();
                    ISingleAxis a = c.GetAxis(AxisRef.A);
                    while (a.MotionStat != 0 &&
                           stopTime.ElapsedMilliseconds < 5000)
                        Thread.Sleep(20);
                    if (a.MotionStat == 0)
                    {
                        AAMotionAPI.MotorOff(c, AxisRef.A);
                        Console.WriteLine("stopped, motor off");
                    }
                    else
                        Console.WriteLine("stop unconfirmed; check the axis");
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("failed, verify axis state: " + ex.Message);
            }
            finally
            {
                AAMotionAPI.Disconnect(c);
            }
        }
    }
}

    // 记录一次运动 / Record a move
// 顺序：连接 → 换相 → 使能 → 本次操作 → 确认结果 → 清理与断开
#include "aamotion.h"
#include <chrono>
#include <cmath>
#include <cstdio>
#include <filesystem>
#include <fstream>
#include <iostream>
#include <string>
#include <thread>
#include <cstdlib>
#include <vector>
// 控制器 IP（出厂默认 172.1.1.101）；用仿真器时改成 firmware_sim.exe 的完整路径
static const std::string Target = "172.1.1.101";
static const char* Step = "4/6 RecordMove";
static const char* ResultStep = "5/6 Result";
static const int A = AAMOTION_AXIS_REF_A;
static bool motorOnRequested = false;

static void Log(const char* step, const std::string& text)
{
    std::printf("[%s] %s\n", step, text.c_str());
}

static void OnError(int code, const char* message, const char* exception,
                    void* user)
{
    Log("ERR", std::to_string(code) + ": " + (message ? message : "") + " | " +
        (exception ? exception : ""));
}

// 读一个数值；失败时返回 NAN，不能当作 0 使用
static double Read(aamotion_controller h, int axis, int keyword, int index = 0)
{
    double value = NAN;
    aamotion_get_value(h, axis, keyword, index, &value);
    return value;
}

static std::string Num(double v)
{
    char text[32];
    std::snprintf(text, sizeof text, "%.10g", v);
    return text;
}

// 使能与控制器故障状态，用于说明失败原因
static std::string State(aamotion_controller h, int axis)
{
    double fault = Read(h, axis, AAMOTION_KEYWORD_CON_FLT);
    char text[256] = "";
    if (!std::isnan(fault))
        aamotion_get_con_flt_des((int)fault, text, sizeof text);
    return "MotorOn=" + Num(Read(h, axis, AAMOTION_KEYWORD_MOTOR_ON)) +
           ", ConFlt=" + Num(fault) + " " + text;
}

static long long Ms(std::chrono::steady_clock::time_point since)
{
    using namespace std::chrono;
    return duration_cast<milliseconds>(steady_clock::now() - since).count();
}

static void SleepMs(int ms)
{
    std::this_thread::sleep_for(std::chrono::milliseconds(ms));
}

static bool IsSimulator(const std::string& target)
{
    return target.size() > 4 && target.compare(target.size() - 4, 4, ".exe") == 0;
}

static int Operate(aamotion_controller h)
{
    // 1. 连接：Target 是 IP 时连接控制器；以 .exe 结尾时连接仿真器
    Log("1/6 Connect", Target);
    bool simulator = IsSimulator(Target);
    bool connected;
    if (!simulator)
        connected = aamotion_connect(h, Target.c_str(), true, false);
    else if (std::filesystem::exists(std::filesystem::u8path(Target)))
    {
        aamotion_init(h);   // 仿真器需要先启动 AACommServer
        connected = aamotion_connect_simulation(h, Target.c_str(),
            nullptr, nullptr, nullptr);
    }
    else
    {
        Log("1/6 Connect", "simulator not found; edit Target");
        return 2;
    }
    if (!connected)
    {
        Log("1/6 Connect", "failed; check Target and the cable");
        return 2;
    }

    // 2. 换相：仿真用虚拟电机。真实电机才需要换相，等待有上限
    if (simulator)
        aamotion_set_value(h, A, AAMOTION_KEYWORD_MOTOR_TYPE, 0, 5);
    else if (!aamotion_is_auto_phase(h, A))
    {
        Log("2/6 AutoPhase", "requested");
        aamotion_auto_phase(h, A);
        auto phaseTime = std::chrono::steady_clock::now();
        while (!aamotion_is_auto_phase(h, A))
        {
            if (Ms(phaseTime) >= 10000)
            {
                Log("2/6 AutoPhase", "timed out; motor stays off");
                return 3;
            }
            SleepMs(20);
        }
    }
    Log("2/6 AutoPhase", simulator ? "virtual motor" : "commutated");

    // 3. 使能：清理阶段也会处理结果不确定的使能
    //    控制器回复 ERR 时错误回调会打印错误码；超时等失败没有回调，
    //    所以失败时再读一次使能与故障状态
    motorOnRequested = true;
    if (!aamotion_motor_on(h, A))
    {
        Log("3/6 MotorOn", "returned false; " + State(h, A));
        return 4;
    }
    if (Read(h, A, AAMOTION_KEYWORD_MOTOR_ON) != 1)
    {
        Log("3/6 MotorOn", "accepted but not on; " + State(h, A));
        return 4;
    }
    Log("3/6 MotorOn", "on");

    // 4. 本次操作
    // 状态 2 表示预触发数据已就绪。
    char state[32] = {};
    if (!aamotion_send_command_string(h, "ARecStat[1]", state,
                                      sizeof state))
    {
        Log(Step, "ARecStat read failed; see any ERR line above");
        return 6;
    }
    int initial = std::atoi(state);
    if (initial >= 1 && initial <= 3)
    {
        Log(Step, "recorder busy: ARecStat=" + std::to_string(initial));
        return 6;
    }

    bool moved = false;
    std::thread mover([h, &moved] {
        auto start = std::chrono::steady_clock::now();
        char reply[32] = {};
        while (Ms(start) < 2000)
        {
            if (aamotion_send_command_string(h, "ARecStat[1]", reply,
                                            sizeof reply) &&
                std::atoi(reply) == 2)
            {
                moved = aamotion_move_relative(h, A, 5000, 5000,
                                               50000, 50000);
                return;
            }
            SleepMs(10);
        }
    });
    // 记录 1 s、每 1 ms 一点、触发前保留 10 %；
    // 类型 5：AMotionStat 从 0 升为非零，即 A 轴开始运动
    bool ok = aamotion_data_rec_record(h, "APos,APosErr", 1.0, 1.0, 10,
                                       5, "AMotionStat", 0, -1,
                                       false, 5000);
    mover.join();
    if (!ok || !moved)
    {
        Log(Step, "recording failed; see the ERR line");
        return 6;
    }
    int length = 0, gap = 0, vectors = 0, samples = 0;
    bool hasData = aamotion_data_rec_info(h, &length, &gap,
                                          &vectors, &samples);
    if (!hasData || vectors != 2 || samples <= 0)
    {
        Log(Step, "no recorded data");
        return 6;
    }
    std::vector<double> time(samples), recordedPos(samples);
    std::vector<double> error(samples);
    int times = 0, positions = 0, errors = 0;
    if (!aamotion_data_rec_time(h, time.data(), samples, &times) ||
        !aamotion_data_rec_vector_values(h, 0, recordedPos.data(),
                                         samples, &positions) ||
        !aamotion_data_rec_vector_values(h, 1, error.data(),
                                         samples, &errors) ||
        times != samples || positions != samples || errors != samples)
    {
        Log(Step, "reading the recording failed");
        return 6;
    }

    std::ofstream csv("move.csv");
    csv << "time_s,APos,APosErr\n";
    for (int i = 0; i < samples; i++)
        csv << time[i] << ',' << recordedPos[i] << ','
            << error[i] << '\n';
    csv.close();
    if (!csv)
    {
        Log(Step, "writing move.csv failed");
        return 6;
    }
    Log(Step, "saved move.csv");

    // 5. 确认结果：命令被接受不等于动作完成
    auto motionTime = std::chrono::steady_clock::now();
    // 4：整定到位
    const double reached = AAMOTION_IN_TARGET_STAT_TARGET_REACHED;
    while (Read(h, A, AAMOTION_KEYWORD_IN_TARGET_STAT) != reached)
    {
        if (Ms(motionTime) >= 10000)
        {
            aamotion_stop(h, A);
            Log(ResultStep, "settle timeout; stop requested");
            return 6;
        }
        SleepMs(20);
    }
    double pos = Read(h, A, AAMOTION_KEYWORD_POS);
    Log(ResultStep, "settled: Pos=" + Num(pos));
    return 0;
}

static int Run()
{
    std::printf("Axis A will move. Check travel and units.\n");
    std::printf("Type RUN to continue:\n");
    std::string answer;
    if (!std::getline(std::cin, answer) || answer != "RUN") return 1;
    aamotion_controller h = aamotion_initialize(AAMOTION_CONTROLLER_TYPE_AGM800);
    if (!h)
    {
        Log("1/6 Connect", "initialize failed; AACommServerNative missing?");
        return 2;
    }
    aamotion_set_error_occurred_handler(h, OnError, nullptr);
    int result = Operate(h);
    // 6. 清理：先确认停止再断使能；断开通信不能证明设备已停止
    if (motorOnRequested)
    {
        aamotion_stop(h, A);
        auto stopTime = std::chrono::steady_clock::now();
        while (Read(h, A, AAMOTION_KEYWORD_MOTION_STAT) != 0 &&
               Ms(stopTime) < 5000)
            SleepMs(20);
        if (Read(h, A, AAMOTION_KEYWORD_MOTION_STAT) == 0)
        {
            aamotion_motor_off(h, A);
            Log("6/6 Cleanup", "stopped, motor off");
        }
        else
            Log("6/6 Cleanup", "stop unconfirmed; check the axis");
    }
    aamotion_disconnect(h);   // 同时释放句柄
    return result;
}

int main()
{
    int result = Run();
    std::printf("Exit code %d.\n", result);
    return result;
}

    // 慢速记录 2 秒 / Log slowly for 2 s
// 顺序：连接 → 本次操作 → 断开。轴不运动。
#include "aamotion.h"
#include <chrono>
#include <cmath>
#include <cstdio>
#include <filesystem>
#include <fstream>
#include <iostream>
#include <string>
#include <thread>
#include <vector>
// 控制器 IP（出厂默认 172.1.1.101）；用仿真器时改成 firmware_sim.exe 的完整路径
static const std::string Target = "172.1.1.101";
static const char* Step = "2/3 SlowLogger";
static const int A = AAMOTION_AXIS_REF_A;

static void Log(const char* step, const std::string& text)
{
    std::printf("[%s] %s\n", step, text.c_str());
}

static void OnError(int code, const char* message, const char* exception,
                    void* user)
{
    Log("ERR", std::to_string(code) + ": " + (message ? message : "") + " | " +
        (exception ? exception : ""));
}

static void SleepMs(int ms)
{
    std::this_thread::sleep_for(std::chrono::milliseconds(ms));
}

static bool IsSimulator(const std::string& target)
{
    return target.size() > 4 && target.compare(target.size() - 4, 4, ".exe") == 0;
}

static int Operate(aamotion_controller h)
{
    // 1. 连接：Target 是 IP 时连接控制器；以 .exe 结尾时连接仿真器
    Log("1/3 Connect", Target);
    bool simulator = IsSimulator(Target);
    bool connected;
    if (!simulator)
        connected = aamotion_connect(h, Target.c_str(), true, false);
    else if (std::filesystem::exists(std::filesystem::u8path(Target)))
    {
        aamotion_init(h);   // 仿真器需要先启动 AACommServer
        connected = aamotion_connect_simulation(h, Target.c_str(),
            nullptr, nullptr, nullptr);
    }
    else
    {
        Log("1/3 Connect", "simulator not found; edit Target");
        return 2;
    }
    if (!connected)
    {
        Log("1/3 Connect", "failed; check Target and the cable");
        return 2;
    }

    // 2. 本次操作
    // 间隔单位是 16 个采样周期：10 约为 9.8 ms；模式 0：FIFO 满时暂停
    const char* names[] = {"APos", "AVel[1]"};
    if (!aamotion_configure_slow_logger(h, 10, 0, names, 2))
    {
        Log(Step, "configuring the logger failed; see the ERR line");
        return 6;
    }
    aamotion_set_logger_on(h, 1);
    Log(Step, "logging for 2 s");
    SleepMs(2000);
    std::vector<char> raw(1 << 20);   // 取出控制器 FIFO 中的数据
    int size = (int)raw.size();
    bool uploaded = aamotion_logger_upload(h, raw.data(), size);
    aamotion_set_logger_on(h, 0);
    std::string text = uploaded ? raw.data() : "";
    Log(Step, "uploaded " + std::to_string(text.size()) + " chars");
    std::string head = text.substr(0, 60);
    for (char& ch : head)
        if (ch == '\r' || ch == '\n') ch = ' ';
    Log(Step, "first: " + head);
    Log(Step, "lost packets " +
        std::to_string(aamotion_get_logger_lost_packets(h)) +
        ", free space " +
        std::to_string(aamotion_get_logger_free_space(h)));
    return 0;
}

static int Run()
{
    aamotion_controller h = aamotion_initialize(AAMOTION_CONTROLLER_TYPE_AGM800);
    if (!h)
    {
        Log("1/3 Connect", "initialize failed; AACommServerNative missing?");
        return 2;
    }
    aamotion_set_error_occurred_handler(h, OnError, nullptr);
    int result = Operate(h);
    // 3. 断开：同时释放句柄；AACommServer 继续为其他程序服务
    aamotion_disconnect(h);
    Log("3/3 Disconnect", "done");
    return result;
}

int main()
{
    int result = Run();
    std::printf("Exit code %d.\n", result);
    return result;
}

    First motion¶

    • You need a Windows PC and the simulator (or a controller whose axis can safely move 10000 counts)
    • Read first First program
    • You will have a complete program that connects, enables, moves, waits in target, stops and disconnects

    This page is sample 02-first-move; the full projects in three languages are under samples/:

    Language Project Run
    C# samples/csharp/02-FirstMove 02-FirstMove.exe "D:\sim\AGM800\firmware_sim.exe"
    C++ samples/cpp/02-first-move sample02_first_move "D:\sim\AGM800\firmware_sim.exe"
    Python samples/python/02_first_move.py python 02_first_move.py "D:\sim\AGM800\firmware_sim.exe"

    Run it on the simulator first

    The program moves axis A 10000 counts forward from where it is. Before real equipment, make sure that travel is safe and the E-stop works. See the Safety checklist.

    Step 1: connect¶

    As in First program, the argument picks the connection: an IP or an .exe path. A failed connect still disconnects, to release what initialization created.

    MotionController c = AAMotionAPI.Initialize(ControllerType.AGM800);
c.ErrorOccurred += (code, sent, text) => Console.WriteLine($"[ERR {code}] {sent} | {text.Replace("\n", " | ")}");
bool ok;
if (target.EndsWith(".exe", StringComparison.OrdinalIgnoreCase))
{
    c.Init();
    ok = AAMotionAPI.ConnectSimulation(c, target);
}
else
{
    ok = AAMotionAPI.Connect(c, target);
}
Console.WriteLine($"1. connect {target}: {ok}");
if (!ok)
{
    AAMotionAPI.Disconnect(c);
    return 1;
}

    aamotion_controller h = aamotion_initialize(AAMOTION_CONTROLLER_TYPE_AGM800);
if (!h)
{
    std::fprintf(stderr, "initialize failed: is AACommServerNative beside the library?\n");
    return 1;
}
aamotion_set_error_occurred_handler(h, on_error, nullptr);
bool is_sim = target.size() > 4 && (target.compare(target.size() - 4, 4, ".exe") == 0 || target.compare(target.size() - 4, 4, ".EXE") == 0);
bool ok = is_sim ? aamotion_connect_simulation(h, target.c_str(), "AGM800", nullptr, nullptr)
                 : aamotion_connect(h, target.c_str(), false, false);
std::printf("1. connect %s: %s\n", target.c_str(), ok ? "true" : "false");
if (!ok)
{
    aamotion_disconnect(h);
    return 1;
}

    h = am.initialize(am.CONTROLLER_AGM800)
if not h:
    sys.exit("initialize failed: is AACommServerNative beside the library?")
am.lib.aamotion_set_error_occurred_handler(h, on_error, None)
if target.lower().endswith(".exe"):
    ok = am.lib.aamotion_connect_simulation(h, target.encode(), b"AGM800", None, None)
else:
    ok = am.lib.aamotion_connect(h, target.encode(), False, False)
print(f"1. connect {target}: {ok}")
if not ok:
    am.lib.aamotion_disconnect(h)
    sys.exit(1)

    You should see: 1. connect ...: True. On False, follow the "Connect" column of the Diagnosis flow.

    Step 2: commutate before motor-on¶

    Check commutation first; if incomplete, call AutoPhase and wait for success before MotorOn. The simulator uses this order too. Skip repeated commutation only when already confirmed; do not enable or move after a commutation timeout.

    if (!AAMotionAPI.IsAutoPhase(c, AxisRef.A))
{
    AAMotionAPI.AutoPhase(c, AxisRef.A);
    DateTime end = DateTime.UtcNow.AddSeconds(10);
    while (!AAMotionAPI.IsAutoPhase(c, AxisRef.A))
    {
        if (DateTime.UtcNow > end)
        {
            Console.WriteLine($"2. commutation timed out: ComtStatus[1] = {a.ComtStatus[1]}");
            return 3;
        }
        Thread.Sleep(100);
    }
}
Console.WriteLine("2. commutated");
AAMotionAPI.MotorOn(c, AxisRef.A);
Console.WriteLine($"3. motor on: MotorOn = {a.MotorOn}");

    if (!aamotion_is_auto_phase(h, A))
{
    aamotion_auto_phase(h, A);
    int waited = 0;
    while (!aamotion_is_auto_phase(h, A) && waited < 10000)
    {
        sleep_ms(100);
        waited += 100;
    }
}
bool commutated = aamotion_is_auto_phase(h, A);
std::printf("2. commutated: %s\n", commutated ? "true" : "false");
bool enabled = commutated && aamotion_motor_on(h, A);
if (enabled)
    std::printf("3. motor on: MotorOn = %.0f\n", read(h, AAMOTION_KEYWORD_MOTOR_ON));

    if not am.lib.aamotion_is_auto_phase(h, A):
    am.lib.aamotion_auto_phase(h, A)
    end = time.monotonic() + 10
    while not am.lib.aamotion_is_auto_phase(h, A) and time.monotonic() < end:
        time.sleep(0.1)
commutated = am.lib.aamotion_is_auto_phase(h, A)
print("2. commutated:", commutated)
if commutated:
    if not am.lib.aamotion_motor_on(h, A):
        raise RuntimeError("motor-on refused")
    print("3. motor on: MotorOn =", am.get_value(h, A, am.KEYWORD_MOTOR_ON))

    You should see: 2. commutated, followed by 3. motor on: MotorOn = 1.

    Step 3: move and wait in target¶

    C# provides MoveAbsAndWait.

    C provides aamotion_move_abs_and_wait. The example below polls the status itself, to show what the helper waits for.

    // Read live status: 4 means settled; 0 MotionStat means only motion ended.
static bool wait_in_target(aamotion_controller h, int timeout_ms)
{
    const auto deadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(timeout_ms);
    while (std::chrono::steady_clock::now() < deadline)
    {
        double stat = 0;
        if (!aamotion_get_value(h, A, AAMOTION_KEYWORD_IN_TARGET_STAT, 0, &stat))
            return false;
        if (stat == AAMOTION_IN_TARGET_STAT_TARGET_REACHED)
            return true;
        sleep_ms(50);
    }
    return false;
}

    def wait_in_target(h, timeout_s=10.0):
    """Read live status; 4 means settled. The current SDK cache handles old cached states."""
    end = time.monotonic() + timeout_s
    while time.monotonic() < end:
        stat = am.get_value(h, A, am.KEYWORD_IN_TARGET_STAT)
        if stat is None:
            return False
        if stat == am.IN_TARGET_STAT_TARGET_REACHED:
            return True
        time.sleep(0.05)
    return False

    Motion ended versus settled

    MotionStat == 0 means motion ended; InTargetStat == 4 means settled. After Begin, the fast read cache does not answer with a status from before the start. This sample reads the settling status live, with a timeout and read-failure handling.

    double start = a.Pos;
bool arrived = AAMotionAPI.MoveAbsAndWait(c, AxisRef.A, start + Distance, Speed, Accel, Accel, timeout: 10000);
Console.WriteLine($"4. move {start} -> {start + Distance}: in target = {arrived}, Pos = {a.Pos}");
if (!arrived)
{
    AAMotionAPI.Stop(c, AxisRef.A);
    int reason = (int)c.GetValue(AxisRef.A, eKeyword.MotionReason);
    Console.WriteLine($"   stopped; MotionReason = {reason} {EnumPropertyHelper.GetMotionReasonDes(reason)}");
    return 4;
}

    double start = read(h, AAMOTION_KEYWORD_POS);
bool accepted = std::isfinite(start) && aamotion_move_abs(h, A, start + Distance, Speed, Accel, Accel);
bool arrived = accepted && wait_in_target(h, 10000);
std::printf("4. move %.0f -> %.0f: in target = %s, Pos = %.0f\n", start, start + Distance,
            arrived ? "true" : "false", read(h, AAMOTION_KEYWORD_POS));
if (!arrived)
    rc = 4;

    start = am.get_value(h, A, am.KEYWORD_POS)
if start is None:
    raise RuntimeError("could not read the starting position")
if not am.lib.aamotion_move_abs(h, A, start + DISTANCE, SPEED, ACCEL, ACCEL):
    raise RuntimeError("move refused; see the error callback")
arrived = wait_in_target(h)
if not arrived:
    am.lib.aamotion_stop(h, A)
print(f"4. move {start:.0f} -> {start + DISTANCE:.0f}: in target = {arrived}, Pos = {am.get_value(h, A, am.KEYWORD_POS)}")
rc = 0 if arrived else 4

    Acceptance: in target = True corresponds to InTargetStat == 4, satisfying the configured tolerance and settling time. MotionStat == 0 means only motion ended. Observe Pos without requiring exact equality. On failure request Stop and inspect errors, MotionReason and actual axis state.

    Step 4: stop, disable, disconnect¶

    This runs whether the steps before succeeded or not. In a finally, an exception cannot skip it.

    AAMotionAPI.Stop(c, AxisRef.A);       // harmless when already still
AAMotionAPI.MotorOff(c, AxisRef.A);
AAMotionAPI.Disconnect(c);            // also releases the controller object
Console.WriteLine("5. motor off, disconnected");

    aamotion_stop(h, A);          // harmless when already still
aamotion_motor_off(h, A);
aamotion_disconnect(h);       // also frees h
std::printf("5. motor off, disconnected\n");

    am.lib.aamotion_stop(h, A)
am.lib.aamotion_motor_off(h, A)
am.lib.aamotion_disconnect(h)
print("5. motor off, disconnected")

    Whole output¶

    On the simulator the output looks like this (positions depend on where the simulator axis is):

    1. connect D:\sim\AGM800\firmware_sim.exe: True
2. commutated
3. motor on: MotorOn = 1
4. move 0 -> 10000: in target = True, Pos = 10000
5. motor off, disconnected
    Exit code Meaning
    0 all steps succeeded
    1 connect failed
    3 commutation timed out
    4 not in target within 10 s
    5 a command was refused (C# only)

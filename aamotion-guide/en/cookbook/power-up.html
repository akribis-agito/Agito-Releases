    Power-up to the first move¶

    A controller fresh from power-up passes these steps before it can move. The program below checks each result and says which step failed and why.

    Step What When it fails, look at
    1. Connect Connect IP, subnet, model; see Connecting
    2. Commutation AutoPhase + poll IsAutoPhase ComtStatus[1]; moving before commutation gives ERR 31
    3. Motor on MotorOn the ERR number at the error hook; see Error lookup
    4. Homing Home + poll IsHomed / HomingStat negative HomingStat values are in Enums
    5. Move MoveAbsAndWait MotionReason: why the move ended

    Configure axis A homing in PCSuite, then follow the demo run instructions.

    using System;
using System.Diagnostics;
using System.IO;
using System.Threading;
using AAMotion;
// 回零后运动 / Home then move
// 顺序：连接 → 换相 → 使能 → 本次操作 → 确认结果 → 清理与断开
internal static class Program
{
    // 控制器 IP（出厂默认 172.1.1.101）；用仿真器时改成 firmware_sim.exe 的完整路径
    private const string Target = "172.1.1.101";

    private static int Main(string[] args)
    {
        Console.WriteLine("Axis A will move. Check travel and units.");
        Console.WriteLine("Type RUN to continue:");
        if (Console.ReadLine() != "RUN") return 1;
        string connectionTarget = args.Length == 0 ? Target : args[0];
        MotionController c = AAMotionAPI.Initialize(ControllerType.AGM800);
        c.ErrorOccurred += (code, command, message) =>
            Console.WriteLine($"{code}: {command} | {message}");
        bool motorOnRequested = false;
        try
        {
            // 1. 连接：Target 是 IP 时连接控制器；以 .exe 结尾时连接仿真器
            Console.WriteLine(connectionTarget);
            bool connected;
            bool simulation = connectionTarget.EndsWith(".exe",
                StringComparison.OrdinalIgnoreCase);
            if (!simulation)
                connected = AAMotionAPI.Connect(c, connectionTarget);
            else if (File.Exists(connectionTarget))
            {
                c.Init();   // 仿真器需要先启动 AACommServer
                connected = AAMotionAPI.ConnectSimulation(c, connectionTarget);
            }
            else
            {
                Console.WriteLine("simulator not found; edit Target");
                return 2;
            }
            if (!connected)
            {
                Console.WriteLine("failed; check Target and the cable");
                return 2;
            }
            ISingleAxis a = c.GetAxis(AxisRef.A);

            // 仿真用虚拟电机。真实电机才需要换相。
            if (simulation)
                a.MotorType = 5;
            else if (!AAMotionAPI.IsAutoPhase(c, AxisRef.A))
            {
                Console.WriteLine("requested");
                AAMotionAPI.AutoPhase(c, AxisRef.A);
                var phaseTime = Stopwatch.StartNew();
                while (!AAMotionAPI.IsAutoPhase(c, AxisRef.A))
                {
                    if (phaseTime.ElapsedMilliseconds >= 10000)
                    {
                        Console.WriteLine("timed out; motor stays off");
                        return 3;
                    }
                    Thread.Sleep(20);
                }
            }
            if (!simulation) Console.WriteLine("commutated");

            // 3. 使能：清理阶段也会处理结果不确定的使能
            motorOnRequested = true;
            AAMotionAPI.MotorOn(c, AxisRef.A);
            if (a.MotorOn != 1)
            {
                Console.WriteLine("refused; MotorOn=" + a.MotorOn);
                return 4;
            }
            Console.WriteLine("on");

            // 4. 本次操作
            a.Home();
            var homeTime = Stopwatch.StartNew();
            while (!AAMotionAPI.IsHomed(c, AxisRef.A))   // HomingStat == 100
            {
                int stat = (int)c.GetValue(AxisRef.A, eKeyword.HomingStat);
                if (stat < 0)
                {
                    string reason = EnumPropertyHelper.GetHomingStatDes(stat);
                    Console.WriteLine("failed: " + reason);
                    return 6;
                }
                if (homeTime.ElapsedMilliseconds >= 30000)
                {
                    AAMotionAPI.Stop(c, AxisRef.A);
                    Console.WriteLine("homing timeout; Stop requested");
                    return 6;
                }
                Thread.Sleep(100);
            }
            Console.WriteLine($"homed: Pos={a.Pos}");
            if (!a.MoveAbsAndWait(10000, 2000, 10000, 10000,
                                  timeout: 10000))
                return 6;

            // 5. 确认结果：方法返回不等于动作完成
            Console.WriteLine($"settled: Pos={a.Pos}");
            return 0;
        }
        catch (ControllerCommandException ex)
        {
            Console.WriteLine($"{ex.Command} refused: ERR {ex.ErrorCode}");
            return 5;
        }
        catch (Exception ex)
        {
            Console.WriteLine(ex.Message);
            return 5;
        }
        finally
        {
            // 6. 清理：先确认停止再断使能
            //    断开通信不能证明设备已停止
            try
            {
                if (motorOnRequested)
                {
                    AAMotionAPI.Stop(c, AxisRef.A);
                    var stopTime = Stopwatch.StartNew();
                    ISingleAxis a = c.GetAxis(AxisRef.A);
                    while (a.MotionStat != 0 &&
                           stopTime.ElapsedMilliseconds < 5000)
                        Thread.Sleep(20);
                    if (a.MotionStat == 0)
                    {
                        AAMotionAPI.MotorOff(c, AxisRef.A);
                        Console.WriteLine("stopped, motor off");
                    }
                    else
                        Console.WriteLine("stop unconfirmed; check the axis");
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("failed, verify axis state: " + ex.Message);
            }
            finally
            {
                AAMotionAPI.Disconnect(c);
            }
        }
    }
}

    // 回零后运动 / Home then move
// 顺序：连接 → 换相 → 使能 → 本次操作 → 确认结果 → 清理与断开
#include "aamotion.h"
#include <chrono>
#include <cmath>
#include <cstdio>
#include <filesystem>
#include <fstream>
#include <iostream>
#include <string>
#include <thread>
// 控制器 IP（出厂默认 172.1.1.101）；用仿真器时改成 firmware_sim.exe 的完整路径
static const std::string Target = "172.1.1.101";
static const char* Step = "4/6 PowerUpMove";
static const char* ResultStep = "5/6 Result";
static const int A = AAMOTION_AXIS_REF_A;
static bool motorOnRequested = false;

static void Log(const char* step, const std::string& text)
{
    std::printf("[%s] %s\n", step, text.c_str());
}

static void OnError(int code, const char* message, const char* exception,
                    void* user)
{
    Log("ERR", std::to_string(code) + ": " + (message ? message : "") + " | " +
        (exception ? exception : ""));
}

// 读一个数值；失败时返回 NAN，不能当作 0 使用
static double Read(aamotion_controller h, int axis, int keyword, int index = 0)
{
    double value = NAN;
    aamotion_get_value(h, axis, keyword, index, &value);
    return value;
}

static std::string Num(double v)
{
    char text[32];
    std::snprintf(text, sizeof text, "%.10g", v);
    return text;
}

// 使能与控制器故障状态，用于说明失败原因
static std::string State(aamotion_controller h, int axis)
{
    double fault = Read(h, axis, AAMOTION_KEYWORD_CON_FLT);
    char text[256] = "";
    if (!std::isnan(fault))
        aamotion_get_con_flt_des((int)fault, text, sizeof text);
    return "MotorOn=" + Num(Read(h, axis, AAMOTION_KEYWORD_MOTOR_ON)) +
           ", ConFlt=" + Num(fault) + " " + text;
}

static long long Ms(std::chrono::steady_clock::time_point since)
{
    using namespace std::chrono;
    return duration_cast<milliseconds>(steady_clock::now() - since).count();
}

static void SleepMs(int ms)
{
    std::this_thread::sleep_for(std::chrono::milliseconds(ms));
}

static bool IsSimulator(const std::string& target)
{
    return target.size() > 4 && target.compare(target.size() - 4, 4, ".exe") == 0;
}

static int Operate(aamotion_controller h)
{
    // 1. 连接：Target 是 IP 时连接控制器；以 .exe 结尾时连接仿真器
    Log("1/6 Connect", Target);
    bool simulator = IsSimulator(Target);
    bool connected;
    if (!simulator)
        connected = aamotion_connect(h, Target.c_str(), true, false);
    else if (std::filesystem::exists(std::filesystem::u8path(Target)))
    {
        aamotion_init(h);   // 仿真器需要先启动 AACommServer
        connected = aamotion_connect_simulation(h, Target.c_str(),
            nullptr, nullptr, nullptr);
    }
    else
    {
        Log("1/6 Connect", "simulator not found; edit Target");
        return 2;
    }
    if (!connected)
    {
        Log("1/6 Connect", "failed; check Target and the cable");
        return 2;
    }

    // 2. 换相：仿真用虚拟电机。真实电机才需要换相，等待有上限
    if (simulator)
        aamotion_set_value(h, A, AAMOTION_KEYWORD_MOTOR_TYPE, 0, 5);
    else if (!aamotion_is_auto_phase(h, A))
    {
        Log("2/6 AutoPhase", "requested");
        aamotion_auto_phase(h, A);
        auto phaseTime = std::chrono::steady_clock::now();
        while (!aamotion_is_auto_phase(h, A))
        {
            if (Ms(phaseTime) >= 10000)
            {
                Log("2/6 AutoPhase", "timed out; motor stays off");
                return 3;
            }
            SleepMs(20);
        }
    }
    Log("2/6 AutoPhase", simulator ? "virtual motor" : "commutated");

    // 3. 使能：清理阶段也会处理结果不确定的使能
    //    控制器回复 ERR 时错误回调会打印错误码；超时等失败没有回调，
    //    所以失败时再读一次使能与故障状态
    motorOnRequested = true;
    if (!aamotion_motor_on(h, A))
    {
        Log("3/6 MotorOn", "returned false; " + State(h, A));
        return 4;
    }
    if (Read(h, A, AAMOTION_KEYWORD_MOTOR_ON) != 1)
    {
        Log("3/6 MotorOn", "accepted but not on; " + State(h, A));
        return 4;
    }
    Log("3/6 MotorOn", "on");

    // 4. 本次操作
    if (!aamotion_home(h, A))
    {
        Log(Step, "home failed; see any ERR line above");
        return 6;
    }
    auto homeTime = std::chrono::steady_clock::now();
    while (!aamotion_is_homed(h, A))   // HomingStat == 100
    {
        double stat = Read(h, A, AAMOTION_KEYWORD_HOMING_STAT);
        if (stat < 0)
        {
            char text[256] = "";
            aamotion_get_homing_stat_des((int)stat, text, sizeof text);
            Log(ResultStep, std::string("failed: ") + text);
            return 6;
        }
        if (Ms(homeTime) >= 30000)
        {
            aamotion_stop(h, A);
            Log(ResultStep, "homing timeout; stop requested");
            return 6;
        }
        SleepMs(100);
    }
    double homePos = Read(h, A, AAMOTION_KEYWORD_POS);
    Log(ResultStep, "homed: Pos=" + Num(homePos));
    if (!aamotion_move_abs(h, A, 10000, 2000, 10000, 10000))
    {
        Log(Step, "move_abs failed; see any ERR line above");
        return 6;
    }

    // 5. 确认结果：命令被接受不等于动作完成
    auto motionTime = std::chrono::steady_clock::now();
    // 4：整定到位
    const double reached = AAMOTION_IN_TARGET_STAT_TARGET_REACHED;
    while (Read(h, A, AAMOTION_KEYWORD_IN_TARGET_STAT) != reached)
    {
        if (Ms(motionTime) >= 10000)
        {
            aamotion_stop(h, A);
            Log(ResultStep, "settle timeout; stop requested");
            return 6;
        }
        SleepMs(20);
    }
    double pos = Read(h, A, AAMOTION_KEYWORD_POS);
    Log(ResultStep, "settled: Pos=" + Num(pos));
    return 0;
}

static int Run()
{
    std::printf("Axis A will move. Check travel and units.\n");
    std::printf("Type RUN to continue:\n");
    std::string answer;
    if (!std::getline(std::cin, answer) || answer != "RUN") return 1;
    aamotion_controller h = aamotion_initialize(AAMOTION_CONTROLLER_TYPE_AGM800);
    if (!h)
    {
        Log("1/6 Connect", "initialize failed; AACommServerNative missing?");
        return 2;
    }
    aamotion_set_error_occurred_handler(h, OnError, nullptr);
    int result = Operate(h);
    // 6. 清理：先确认停止再断使能；断开通信不能证明设备已停止
    if (motorOnRequested)
    {
        aamotion_stop(h, A);
        auto stopTime = std::chrono::steady_clock::now();
        while (Read(h, A, AAMOTION_KEYWORD_MOTION_STAT) != 0 &&
               Ms(stopTime) < 5000)
            SleepMs(20);
        if (Read(h, A, AAMOTION_KEYWORD_MOTION_STAT) == 0)
        {
            aamotion_motor_off(h, A);
            Log("6/6 Cleanup", "stopped, motor off");
        }
        else
            Log("6/6 Cleanup", "stop unconfirmed; check the axis");
    }
    aamotion_disconnect(h);   // 同时释放句柄
    return result;
}

int main()
{
    int result = Run();
    std::printf("Exit code %d.\n", result);
    return result;
}

    When commutation or homing is not needed

    Skip commutation only after IsAutoPhase confirms it is complete. The simulator also commutates before motor-on; with an absolute encoder, or a zero already set with SetPosition, drop the homing step.

    Key points¶

    • Attach the error hook before connecting: the reasons for connect and motor-on failures come through it.
    • Every wait has a timeout: 10 s for commutation, 30 s for homing, 10 s for the move; adjust them to your machine.
    • Disconnect in finally: the program disconnects even when an exception ends it. Disconnecting does not close AACommServer.

    Status value to text¶

    EnumPropertyHelper turns status keyword values into English text; bit masks such as MotionStat join the descriptions of every set bit with commas. Every value is listed in the enum reference.

    1. Connect
    2. Operation: StatusText
    3. Disconnect

    C# samples/single/csharp/02-Errors/StatusText · C++ samples/single/cpp/02-Errors/StatusText · Python samples/single/python/02_status_text.py

    Prerequisites¶

    AGM800; no axis moves. Connects to the controller at 172.1.1.101 by default; without hardware, set Target to the full path of the simulator firmware_sim.exe.

    Confirm the result¶

    Prints the MotionReason and MotionStat values with their descriptions.

    Exit codes: 0 success; 1 not confirmed or not applicable; 2 connection failed; 3 commutation timeout; 4 motor-on refused; 5 command failure or exception; 6 wait or result check failed.

    Run¶

    1. Open samples/single/csharp/AAMotionDemos.sln in Visual Studio 2019 or later, or just this project's StatusText.csproj.
    2. In Program.cs, check Target at the top: the controller factory IP 172.1.1.101 by default. To use the simulator, pass the absolute path to firmware_sim.exe as the first debugging argument.
    3. In Solution Explorer right-click StatusText, choose "Set as Startup Project" and press F5.
    4. Needs the .NET Framework 4.8 developer tools; the first build restores the NuGet package Agito.AAMotion 13.0.0.
    1. Extract the native SDK agito-aamotion-native-sdk-13.0.0.zip.
    2. In Visual Studio 2022, "Open a local folder" on samples/single/cpp, and set AAMOTION_SDK_DIR in CMakePresets.json to the SDK folder.
    3. Open samples/single/cpp/02-Errors/StatusText/main.cpp and check Target at the top.
    4. Pick StatusText.exe as the startup item and press F5. The build copies aamotion.dll and AACommServerNative.exe beside the program.

    Linux: cmake --preset linux-x64, cmake --build out/build/linux-x64, then run out/build/linux-x64/bin/StatusText.

    1. Install a 64-bit Python 3.9 or later.
    2. In samples/single/python, run python get_sdk.py once: it downloads the SDK from the Agito release page, checks its SHA256 and puts aamotion.dll and AACommServerNative beside aamotion.py. Copying them from an extracted SDK works too.
    3. Check TARGET at the top of 02_status_text.py and run python 02_status_text.py, or open the file in VS Code and press F5.

    API: Status value to text

    using System;
using System.IO;
using AAMotion;
// 状态值转成文字 / Status value to text
// 顺序：连接 → 本次操作 → 断开。轴不运动。
internal static class Program
{
    // 控制器 IP（出厂默认 172.1.1.101）；用仿真器时改成 firmware_sim.exe 的完整路径
    private const string Target = "172.1.1.101";

    private static int Main(string[] args)
    {
        string connectionTarget = args.Length == 0 ? Target : args[0];
        MotionController c = AAMotionAPI.Initialize(ControllerType.AGM800);
        c.ErrorOccurred += (code, command, message) =>
            Console.WriteLine($"[ERR] {code}: {command} | {message}");
        try
        {
            // 1. 连接：Target 是 IP 时连接控制器；以 .exe 结尾时连接仿真器
            Console.WriteLine(connectionTarget);
            bool connected;
            bool simulation = connectionTarget.EndsWith(".exe",
                StringComparison.OrdinalIgnoreCase);
            if (!simulation)
                connected = AAMotionAPI.Connect(c, connectionTarget);
            else if (File.Exists(connectionTarget))
            {
                c.Init();   // 仿真器需要先启动 AACommServer
                connected = AAMotionAPI.ConnectSimulation(c, connectionTarget);
            }
            else
            {
                Console.WriteLine("simulator not found; edit Target");
                return 2;
            }
            if (!connected)
            {
                Console.WriteLine("failed; check Target and the cable");
                return 2;
            }

            // 2. 本次操作
            int reason = (int)c.GetValue(AxisRef.A, eKeyword.MotionReason);
            string reasonText = EnumPropertyHelper.GetMotionReasonDes(reason);
            Console.WriteLine($"MotionReason={reason}: {reasonText}");
            // 位掩码类：所有置位的位的说明用逗号连接
            int stat = (int)c.GetValue(AxisRef.A, eKeyword.MotionStat);
            string statText = EnumPropertyHelper.GetMotionStatDes(stat);
            Console.WriteLine($"MotionStat={stat}: {statText}");
            return 0;
        }
        catch (ControllerCommandException ex)
        {
            Console.WriteLine($"{ex.Command} refused: ERR {ex.ErrorCode}");
            return 5;
        }
        catch (Exception ex)
        {
            Console.WriteLine(ex.Message);
            return 5;
        }
        finally
        {
            // 3. 断开：AACommServer 继续为其他程序服务
            AAMotionAPI.Disconnect(c);
            Console.WriteLine("done");
        }
    }
}

    // 状态值转成文字 / Status value to text
// 顺序：连接 → 本次操作 → 断开。轴不运动。
#include "aamotion.h"
#include <chrono>
#include <cmath>
#include <cstdio>
#include <filesystem>
#include <fstream>
#include <iostream>
#include <string>
#include <thread>
// 控制器 IP（出厂默认 172.1.1.101）；用仿真器时改成 firmware_sim.exe 的完整路径
static const std::string Target = "172.1.1.101";
static const char* Step = "2/3 StatusText";
static const int A = AAMOTION_AXIS_REF_A;

static void Log(const char* step, const std::string& text)
{
    std::printf("[%s] %s\n", step, text.c_str());
}

static void OnError(int code, const char* message, const char* exception,
                    void* user)
{
    Log("ERR", std::to_string(code) + ": " + (message ? message : "") + " | " +
        (exception ? exception : ""));
}

// 读一个数值；失败时返回 NAN，不能当作 0 使用
static double Read(aamotion_controller h, int axis, int keyword, int index = 0)
{
    double value = NAN;
    aamotion_get_value(h, axis, keyword, index, &value);
    return value;
}

static std::string Num(double v)
{
    char text[32];
    std::snprintf(text, sizeof text, "%.10g", v);
    return text;
}

static bool IsSimulator(const std::string& target)
{
    return target.size() > 4 && target.compare(target.size() - 4, 4, ".exe") == 0;
}

static int Operate(aamotion_controller h)
{
    // 1. 连接：Target 是 IP 时连接控制器；以 .exe 结尾时连接仿真器
    Log("1/3 Connect", Target);
    bool simulator = IsSimulator(Target);
    bool connected;
    if (!simulator)
        connected = aamotion_connect(h, Target.c_str(), true, false);
    else if (std::filesystem::exists(std::filesystem::u8path(Target)))
    {
        aamotion_init(h);   // 仿真器需要先启动 AACommServer
        connected = aamotion_connect_simulation(h, Target.c_str(),
            nullptr, nullptr, nullptr);
    }
    else
    {
        Log("1/3 Connect", "simulator not found; edit Target");
        return 2;
    }
    if (!connected)
    {
        Log("1/3 Connect", "failed; check Target and the cable");
        return 2;
    }

    // 2. 本次操作
    double reason = Read(h, A, AAMOTION_KEYWORD_MOTION_REASON);
    double stat = Read(h, A, AAMOTION_KEYWORD_MOTION_STAT);
    if (std::isnan(reason) || std::isnan(stat))
    {
        Log(Step, "read failed; see the ERR line");
        return 6;
    }
    char text[512] = "";
    aamotion_get_motion_reason_des((int)reason, text, sizeof text);
    Log(Step, "MotionReason=" + Num(reason) + ": " + text);
    // 位掩码类：所有置位的位的说明用逗号连接
    aamotion_get_motion_stat_des((int)stat, text, sizeof text);
    Log(Step, "MotionStat=" + Num(stat) + ": " + text);
    return 0;
}

static int Run()
{
    aamotion_controller h = aamotion_initialize(AAMOTION_CONTROLLER_TYPE_AGM800);
    if (!h)
    {
        Log("1/3 Connect", "initialize failed; AACommServerNative missing?");
        return 2;
    }
    aamotion_set_error_occurred_handler(h, OnError, nullptr);
    int result = Operate(h);
    // 3. 断开：同时释放句柄；AACommServer 继续为其他程序服务
    aamotion_disconnect(h);
    Log("3/3 Disconnect", "done");
    return result;
}

int main()
{
    int result = Run();
    std::printf("Exit code %d.\n", result);
    return result;
}

    """状态值转成文字 / Status value to text
顺序：连接 → 本次操作 → 断开。轴不运动。
"""
import os
import sys
import aamotion as am

# 控制器 IP（出厂默认 172.1.1.101）；用仿真器时改成 firmware_sim.exe 的完整路径
TARGET = "172.1.1.101"
STEP = "2/3 StatusText"
A = am.AXIS_REF_A


def log(step, text):
    print(f"[{step}] {text}")


def on_error(code, message, exception):
    log("ERR", f"{code}: {message} | {exception}")


def operate(h):
    # 1. 连接：TARGET 是 IP 时连接控制器；以 .exe 结尾时连接仿真器
    log("1/3 Connect", TARGET)
    if not TARGET.lower().endswith(".exe"):
        connected = am.connect(h, TARGET, True, False)
    elif os.path.isfile(TARGET):
        am.init(h)   # 仿真器需要先启动 AACommServer
        connected = am.connect_simulation(h, TARGET, None, None, None)
    else:
        log("1/3 Connect", "simulator not found; edit TARGET")
        return 2
    if not connected:
        log("1/3 Connect", "failed; check TARGET and the cable")
        return 2

    # 2. 本次操作
    reason = am.get_value(h, A, am.KEYWORD_MOTION_REASON, 0)
    stat = am.get_value(h, A, am.KEYWORD_MOTION_STAT, 0)
    if reason is None or stat is None:
        log(STEP, "read failed; see the ERR line")
        return 6
    text = am.get_motion_reason_des(int(reason))
    log(STEP, f"MotionReason={reason:g}: {text}")
    # 位掩码类：所有置位的位的说明用逗号连接
    text = am.get_motion_stat_des(int(stat))
    log(STEP, f"MotionStat={stat:g}: {text}")
    return 0


def run():
    h = am.initialize(am.CONTROLLER_TYPE_AGM800)
    if not h:
        log("1/3 Connect", "initialize failed; AACommServerNative missing?")
        return 2
    am.set_error_occurred_handler(h, on_error)
    try:
        return operate(h)
    except Exception as ex:
        log("Failed", str(ex))
        return 5
    finally:
        # 3. 断开：同时释放句柄；AACommServer 继续为其他程序服务
        am.disconnect(h)
        log("3/3 Disconnect", "done")


def main():
    # 输出被重定向时，编码不了的字符（如中文路径）转义输出，不报错；
    # IDLE、Jupyter 的输出没有 reconfigure
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(errors="backslashreplace")
    result = run()
    print(f"Exit code {result}.")
    return result


if __name__ == "__main__":
    sys.exit(main())

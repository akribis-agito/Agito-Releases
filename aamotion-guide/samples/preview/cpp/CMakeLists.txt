cmake_minimum_required(VERSION 3.20)
project(AAMotionPreviewDemos LANGUAGES CXX)

set(AAMOTION_SDK_DIR "" CACHE PATH "Extracted Native preview SDK")
if(NOT AAMOTION_SDK_DIR OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Set AAMOTION_SDK_DIR and configure a 64-bit build")
endif()

if(WIN32)
  set(rid win-x64)
  set(link_library aamotion.lib)
  set(runtime aamotion.dll AACommServerNative.exe)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(rid linux-x64)
  set(link_library libaamotion.so)
  set(runtime libaamotion.so AACommServerNative)
else()
  message(FATAL_ERROR "These examples are configured for Windows x64 and Linux x64")
endif()

foreach(demo IN ITEMS pvaj gearing stall_home)
  add_executable(${demo} ${demo}.cpp)
  target_compile_features(${demo} PRIVATE cxx_std_17)
  target_include_directories(${demo} PRIVATE "${AAMOTION_SDK_DIR}/include")
  target_link_libraries(${demo} PRIVATE "${AAMOTION_SDK_DIR}/${rid}/${link_library}")
  set_target_properties(${demo} PROPERTIES BUILD_RPATH "$ORIGIN")
  foreach(file IN LISTS runtime)
    add_custom_command(TARGET ${demo} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${AAMOTION_SDK_DIR}/${rid}/${file}" "$<TARGET_FILE_DIR:${demo}>")
  endforeach()
endforeach()

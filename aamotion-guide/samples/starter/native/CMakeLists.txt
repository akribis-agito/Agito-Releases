cmake_minimum_required(VERSION 3.20)
project(AAMotionStarter C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
option(STARTER_CPP "Compile the C ABI example as C++" OFF)
if(STARTER_CPP)
  set_source_files_properties(main.c PROPERTIES LANGUAGE CXX)
endif()
add_executable(starter_validate main.c)
target_compile_definitions(starter_validate PRIVATE WB_VALIDATE_ONLY)
set(AAMOTION_SDK_DIR "" CACHE PATH "Extracted native SDK folder")
if(NOT AAMOTION_SDK_DIR)
  message(STATUS "Only starter_validate is built. Set AAMOTION_SDK_DIR for the SDK executable.")
  return()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The native SDK needs a 64-bit process")
endif()
if(WIN32)
  set(RID win-x64)
  set(NATIVE_LIB aamotion.lib)
elseif(APPLE)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(RID osx-arm64)
  else()
    set(RID osx-x64)
  endif()
  set(NATIVE_LIB libaamotion.dylib)
else()
  set(RID linux-x64)
  set(NATIVE_LIB libaamotion.so)
endif()
add_executable(starter main.c)
if(UNIX AND NOT APPLE)
  target_link_libraries(starter PRIVATE m)
endif()
target_include_directories(starter PRIVATE "${AAMOTION_SDK_DIR}/include")
target_link_libraries(starter PRIVATE "${AAMOTION_SDK_DIR}/${RID}/${NATIVE_LIB}")
set_target_properties(starter PROPERTIES BUILD_RPATH "$ORIGIN;@loader_path")
add_custom_command(TARGET starter POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${AAMOTION_SDK_DIR}/${RID}" $<TARGET_FILE_DIR:starter>)
